## Import

```tsx
import { Link } from '@moul-dev/ui';
```

## Usage

Here is a basic example of how to use the `Link` component:

```tsx
import { Link } from '@moul-dev/ui';

export default function Example() {
  return (
    <Link>
      {/* Component content */}
    </Link>
  );
}
```
