# Table

The `Table` component renders accessible, keyboard-navigable data tables with built-in sticky headers, interactive column sorting with visual direction indicators, single and multiple row selection, and customizable loading and empty state slots.

## Column Sorting

Set `allowsSorting` on any `<Column>` to make it interactive. The table automatically renders directional sort indicator icons (ascending, descending, or unsorted):

```tsx
import { Table, TableHeader, Column, TableBody, Row, Cell } from '@moul-dev/ui';
import { useState } from 'react';

export default function SortableExample() {
  const [sortDescriptor, setSortDescriptor] = useState({
    column: 'name',
    direction: 'ascending',
  });

  const [items, setItems] = useState([
    { id: '1', name: 'Auth Gateway', role: 'Security', latency: '24ms' },
    { id: '2', name: 'Database Cluster', role: 'Storage', latency: '4ms' },
    { id: '3', name: 'Edge CDN', role: 'Networking', latency: '142ms' },
  ]);

  const handleSortChange = (descriptor) => {
    setSortDescriptor(descriptor);
    setItems((prev) =>
      [...prev].sort((a, b) => {
        const first = a[descriptor.column];
        const second = b[descriptor.column];
        const cmp = String(first).localeCompare(String(second), undefined, { numeric: true });
        return descriptor.direction === 'descending' ? -cmp : cmp;
      })
    );
  };

  return (
    <Table
      aria-label="Sortable Table"
      sortDescriptor={sortDescriptor}
      onSortChange={handleSortChange}
    >
      <TableHeader>
        <Column id="name" allowsSorting isRowHeader>
          Service Name
        </Column>
        <Column id="role" allowsSorting>
          Role
        </Column>
        <Column id="latency" allowsSorting>
          Latency
        </Column>
      </TableHeader>
      <TableBody items={items}>
        {(item) => (
          <Row key={item.id} id={item.id}>
            <Cell>{item.name}</Cell>
            <Cell>{item.role}</Cell>
            <Cell>{item.latency}</Cell>
          </Row>
        )}
      </TableBody>
    </Table>
  );
}
```

## Row Selection

Enable `selectionMode="single"` or `selectionMode="multiple"` for interactive row selection with full keyboard navigation (arrows, spacebar):

```tsx
import { Table, TableHeader, Column, TableBody, Row, Cell } from '@moul-dev/ui';
import { useState } from 'react';

export default function SelectionExample() {
  const [selectedKeys, setSelectedKeys] = useState(new Set(['1']));

  return (
    <Table
      aria-label="Selectable Table"
      selectionMode="multiple"
      selectedKeys={selectedKeys}
      onSelectionChange={setSelectedKeys}
    >
      <TableHeader>
        <Column isRowHeader>Name</Column>
        <Column>Role</Column>
      </TableHeader>
      <TableBody>
        <Row id="1">
          <Cell>Auth Gateway</Cell>
          <Cell>Security</Cell>
        </Row>
        <Row id="2">
          <Cell>PostgreSQL</Cell>
          <Cell>Database</Cell>
        </Row>
      </TableBody>
    </Table>
  );
}
```

## Sticky Header

Pass `stickyHeader` on `<Table>` (or `sticky` on `<TableHeader>`) to keep column headers pinned while scrolling through data:

```tsx
<div style={{ maxHeight: '280px', overflowY: 'auto' }}>
  <Table aria-label="Telemetry Logs" stickyHeader>
    <TableHeader>
      <Column isRowHeader>Timestamp</Column>
      <Column>Level</Column>
      <Column>Message</Column>
    </TableHeader>
    <TableBody items={logs}>
      {(log) => (
        <Row key={log.id}>
          <Cell>{log.timestamp}</Cell>
          <Cell>{log.level}</Cell>
          <Cell>{log.message}</Cell>
        </Row>
      )}
    </TableBody>
  </Table>
</div>
```

## Loading and Empty States

Configure `isLoading` or `emptyState` slots to provide clear feedback when fetching data or when no rows match query filters:

```tsx
// Loading State
<Table aria-label="Loading Table" isLoading>
  <TableHeader>
    <Column isRowHeader>Name</Column>
    <Column>Status</Column>
  </TableHeader>
  <TableBody items={[]} />
</Table>

// Empty State Slot
<Table
  aria-label="Empty Table"
  emptyState={
    <div className="py-8 text-center text-neutral-500">
      <p className="text-sm font-medium">No results found</p>
      <p className="text-xs mt-1">Try adjusting your search criteria.</p>
    </div>
  }
>
  <TableHeader>
    <Column isRowHeader>Name</Column>
    <Column>Status</Column>
  </TableHeader>
  <TableBody items={[]} />
</Table>
```
