## Import

```tsx
import { ButtonGroup } from '@moul-dev/ui';
```

## Usage

Here is a basic example of how to use the `ButtonGroup` component:

```tsx
import { ButtonGroup } from '@moul-dev/ui';

export default function Example() {
  return (
    <ButtonGroup>
      {/* Component content */}
    </ButtonGroup>
  );
}
```
