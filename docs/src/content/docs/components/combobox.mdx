## Multi-select with TagGroup

You can build a multi-select ComboBox by managing the selection state in a list/set, filtering out the selected items from the dropdown, and rendering them as a `TagGroup` below the ComboBox.

<ComponentPreview>
  <moul-combobox placeholder="Select items..."><moul-combobox-item value="one">Item 1</moul-combobox-item></moul-combobox>
</ComponentPreview>

```tsx



export default function Example() {
  const provinces = [
    { id: 'phnom-penh', name: 'Phnom Penh' },
    { id: 'siem-reap', name: 'Siem Reap' },
    { id: 'battambang', name: 'Battambang' },
    { id: 'sihanoukville', name: 'Sihanoukville' },
    { id: 'kampot', name: 'Kampot' },
    { id: 'kandal', name: 'Kandal' },
    { id: 'kampong-cham', name: 'Kampong Cham' },
    { id: 'koh-kong', name: 'Koh Kong' },
    { id: 'kep', name: 'Kep' },
  ];

  const [selectedKeys, setSelectedKeys] = useState<Set<any>>(new Set());
  const [inputValue, setInputValue] = useState('');

  const availableProvinces = provinces.filter(
    (p) => !selectedKeys.has(p.id)
  );

  const handleSelectionChange = (key: any) => {
    if (key) {
      setSelectedKeys((prev) => {
        const next = new Set(prev);
        next.add(key);
        return next;
      });
      setInputValue('');
    }
  };

  const handleRemove = (keys: Set<any>) => {
    setSelectedKeys((prev) => {
      const next = new Set(prev);
      for (const k of keys) {
        next.delete(k);
      }
      return next;
    });
  };

  return (
    <div class="w-full max-w-sm flex flex-col gap-4">
      <moul-combobox
        label="Cambodian Provinces"
        placeholder="Select a province"
        inputValue={inputValue}
        onInputChange={setInputValue}
        onSelectionChange={handleSelectionChange}
        selected-key={null}
      >
        {availableProvinces.map((province) => (
          <moul-combobox-item key={province.id} id={province.id}>
            {province.name}
          </moul-combobox-item>
        ))}
      </moul-combobox>

      {selectedKeys.size > 0 && (
        <moul-tag-group
          label="Selected Provinces"
          onRemove={handleRemove}
          variant="primary"
        >
          {[...selectedKeys].map((key) => {
            const province = provinces.find((p) => p.id === key);
            return (
              <moul-tag key={key} id={key}>
                {province?.name || key}
              </moul-tag>
            );
          })}
        </moul-tag-group>
      )}
    </div>
  );
}
```

## Item Actions

Use the `onAction` prop on `<moul-combobox-item>` to perform custom actions when an item is pressed (such as dynamic "Create" actions). Custom action items bypass selection and will not populate the input.

Pair this with `allowsEmptyCollection` so the menu stays open when typing a query with no existing matches.

You can also pass `onAction` to `<moul-combobox>` to be notified when any option is activated while preserving normal value selection.

<ComponentPreview>
  <moul-combobox placeholder="Action search..."><moul-combobox-item value="save">Save</moul-combobox-item></moul-combobox>
</ComponentPreview>

```tsx



export default function Example() {
  const [items, setItems] = useState([
    { id: 'react', name: 'React' },
    { id: 'vue', name: 'Vue' },
    { id: 'svelte', name: 'Svelte' },
  ]);
  const [inputValue, setInputValue] = useState('');

  const isExisting = items.some(
    (item) => item.name.toLowerCase() === inputValue.trim().toLowerCase()
  );

  const handleCreate = (name: string) => {
    const trimmed = name.trim();
    if (!trimmed) return;
    const id = trimmed.toLowerCase().replace(/\s+/g, '-');
    setItems((prev) => [...prev, { id, name: trimmed }]);
    setInputValue('');
  };

  return (
    <moul-combobox
      label="Frontend Framework"
      placeholder="Type or select..."
      allowsEmptyCollection
      inputValue={inputValue}
      onInputChange={setInputValue}
      onAction={(key) => console.log('Activated item key:', key)}
    >
      {inputValue.trim().length > 0 && !isExisting && (
        <moul-combobox-item
          id="create-new"
          onAction={() => handleCreate(inputValue)}
        >
          {`Create "${inputValue.trim()}"`}
        </moul-combobox-item>
      )}
      {items.map((item) => (
        <moul-combobox-item key={item.id} id={item.id}>
          {item.name}
        </moul-combobox-item>
      ))}
    </moul-combobox>
  );
}
```
