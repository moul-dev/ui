import ComponentPreview from '@/components/ComponentPreview.astro';

`<moul-text-field>` wraps a native input with a label, helper text and an animated error message. It participates in forms through `ElementInternals`.

<ComponentPreview>
<div class="demo-stack" style="width: min(100%, 22rem)">
  <moul-text-field label="Project name" placeholder="moul-portal" description="Lowercase letters, numbers and dashes."></moul-text-field>
  <moul-text-field label="Email" type="email" value="alex@example" error-message="Enter a complete email address." is-required></moul-text-field>
  <moul-text-field label="API key" value="sk_live_••••" is-readonly></moul-text-field>
  <moul-text-field label="Disabled" placeholder="Unavailable" is-disabled></moul-text-field>
</div>
</ComponentPreview>

| `type` | `type` | `'text' \| 'email' \| 'password' \| 'tel' \| 'url'` | `'text'` |
| `placeholder` | `placeholder` | `string` | `''` |
| `label` | `label` | `string` | `''` |
| `description` | `description` | `string` | `''` |
| `error-message` | `errorMessage` | `string` | `''` |
| `is-disabled` | `isDisabled` | `boolean` | `false` |
| `is-required` | `isRequired` | `boolean` | `false` |
| `is-readonly` | `isReadonly` | `boolean` | `false` |
