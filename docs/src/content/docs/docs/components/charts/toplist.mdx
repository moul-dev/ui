import ComponentPreview from '@/components/ComponentPreview.astro';

`<moul-top-list>` ranks items by value, scaling each bar against the largest.

<ComponentPreview align="stretch">
<moul-top-list show-rank multi-color data-props='{"data":[{"label":"Chrome","value":6420},{"label":"Safari","value":3810},{"label":"Firefox","value":1290},{"label":"Edge","value":980},{"label":"Arc","value":410}]}'></moul-top-list>
</ComponentPreview>

```html
<moul-top-list show-rank multi-color></moul-top-list>
```

```js
const topList = document.querySelector('moul-top-list')
topList.data = [
  {
    "label": "Chrome",
    "value": 6420
  },
```
