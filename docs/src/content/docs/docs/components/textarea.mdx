title: TextArea

import ComponentPreview from '@/components/ComponentPreview.astro';

`<moul-text-area>` accepts longer text. Set `auto-resize` to let it grow as the user types.

<ComponentPreview>
<div class="demo-stack" style="width: min(100%, 26rem)">
  <moul-text-area label="Release notes" placeholder="Summarize what changed…" rows="3" auto-resize description="Markdown is supported."></moul-text-area>
</div>
</ComponentPreview>
