import ComponentPreview from '@/components/ComponentPreview.astro';

`<moul-empty-state>` explains why a view is empty and offers the next step. Its parts rise in with a short stagger.

<ComponentPreview>
<moul-empty-state variant="glass" style="width: min(100%, 28rem)">
  <span slot="icon" style="font-size: 1.75rem">📦</span>
  <span slot="title">No deployments yet</span>
  <span slot="description">Push a commit or trigger a build to see it here.</span>
  <div slot="actions" class="demo-row">
    <moul-button size="sm">New deployment</moul-button>
    <moul-button size="sm" variant="ghost">Read the docs</moul-button>
  </div>
</moul-empty-state>
</ComponentPreview>
